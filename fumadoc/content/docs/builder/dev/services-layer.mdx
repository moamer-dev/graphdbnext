The services layer in Plexus Builder handles business logic, external integrations, and data processing operations. It provides a clean separation between UI components and core functionality, making the system testable and maintainable.

### XML Services

Handles XML parsing, analysis, and transformation:

```typescript
// Parse Service - for CSV and bulk input parsing
export function parseCSVLine(line: string): string[] {
  const result: string[] = []
  let current = ''
  let inQuotes = false
  
  for (let i = 0; i < line.length; i++) {
    const char = line[i]
    const nextChar = line[i + 1]
    
    if (char === '"') {
      if (inQuotes && nextChar === '"') {
        // Escaped quote
        current += '"'
        i++ // Skip next quote
      } else {
        // Toggle quote state
        inQuotes = !inQuotes
      }
    } else if (char === ',' && !inQuotes) {
      // End of field
      result.push(current.trim())
      current = ''
    } else {
      current += char
    }
  }
  
  // Add last field
  result.push(current.trim())
  return result
}

export interface ParsedNode {
  label: string
  type: string
  groupName?: string
}

export function parseBulkNodeInput(input: string): ParsedNode[] {
  const lines = input.split('\n').filter(line => line.trim())
  const nodes: ParsedNode[] = []
  
  for (const line of lines) {
    const parts = parseCSVLine(line)
    
    if (parts.length === 1) {
      // Single word - use as both label and type
      nodes.push({ label: parts[0], type: parts[0] })
    } else if (parts.length >= 2) {
      // label:type or label,type
      const label = parts[0].trim()
      const type = parts[1].trim()
      const groupName = parts[2]?.trim()
      
      nodes.push({ label, type, groupName })
    }
  }
  
  return nodes
}

// XML Analyzer Service
export interface XmlStructureAnalysis {
  elementCounts: Record<string, number>
  attributeCounts: Record<string, number>
  hierarchy: Record<string, string[]>
  depth: number
  totalElements: number
}

export class XmlAnalyzer {
  analyze(document: Document): XmlStructureAnalysis {
    const elementCounts: Record<string, number> = {}
    const attributeCounts: Record<string, number> = {}
    const hierarchy: Record<string, string[]> = {}
    let maxDepth = 0
    let totalElements = 0

    function traverseElement(element: Element, depth: number): void {
      const tagName = element.tagName
      elementCounts[tagName] = (elementCounts[tagName] || 0) + 1
      totalElements++
      maxDepth = Math.max(maxDepth, depth)

      // Count attributes
      for (const attr of Array.from(element.attributes)) {
        attributeCounts[attr.name] = (attributeCounts[attr.name] || 0) + 1
      }

      // Build hierarchy
      const parent = element.parentElement
      if (parent) {
        const parentTag = parent.tagName
        if (!hierarchy[parentTag]) {
          hierarchy[parentTag] = []
        }
        if (!hierarchy[parentTag].includes(tagName)) {
          hierarchy[parentTag].push(tagName)
        }
      }

      // Recurse into children
      for (const child of Array.from(element.children)) {
        if (child.nodeType === Node.ELEMENT_NODE) {
          traverseElement(child as Element, depth + 1)
        }
      }
    }

    traverseElement(document.documentElement, 0)

    return {
      elementCounts,
      attributeCounts,
      hierarchy,
      depth: maxDepth,
      totalElements
    }
  }
}
```

```typescript
/**
 * Import service for handling schema file imports
 */
export class ImportService {
  /**
   * Import schema from a file
   */
  static async importFromFile(file: File): Promise<ImportResult> {
    const text = await file.text()
    const fileExtension = file.name.split('.').pop()?.toLowerCase()

    let nodes: Node[]
    let relationships: Relationship[]
    let metadata = {
      name: 'Imported Schema',
      description: `Imported from ${file.name}`,
      version: '1.0.0'
    }

    if (fileExtension === 'json') {
      // Parse JSON schema
      const schemaJson = JSON.parse(text)

      // Validate schema format
      if (!schemaJson.nodes || !schemaJson.relations) {
        throw new Error('Invalid JSON schema format')
      }

      // Convert to builder format
      const converted = convertSchemaJsonToBuilder(schemaJson)
      nodes = converted.nodes
      relationships = converted.relationships
      metadata = { ...metadata, ...schemaJson.metadata }
    } else if (fileExtension === 'md') {
      // Parse Markdown schema
      const parsed = parseMarkdownSchema(text)
      const converted = convertMarkdownSchemaToBuilder(parsed)
      nodes = converted.nodes
      relationships = converted.relationships
    } else {
      throw new Error(`Unsupported file type: ${fileExtension}`)
    }

    return {
      nodes,
      relationships,
      metadata
    }
  }
}

// Schema Design Service
export interface FindNodesResult {
  source?: Node
  target?: Node
}

export class SchemaDesignService {
  static extractNodeIdFromResult(result: string): string | null {
    const idMatch = result.match(/ID: (node_\d+_[a-z0-9]+)/)
    return idMatch ? idMatch[1] : null
  }

  static extractRelationshipIdFromResult(result: string): string | null {
    const idMatch = result.match(/ID: (rel_\d+_[a-z0-9]+)/)
    return idMatch ? idMatch[1] : null
  }

  static findNodeByLabel(nodes: Node[], label: string): Node | undefined {
    return nodes.find(n => 
      n.label === label || 
      n.type === label.replace(/\s+/g, '')
    )
  }

  static findNodesByMessage(
    nodes: Node[],
    message: string,
    suggestionNodeLabel?: string,
    excludeId?: string
  ): FindNodesResult {
    const messageLower = message.toLowerCase()
    
    const sourceNode = nodes.find(n => 
      n.id !== excludeId && 
      messageLower.includes(n.label.toLowerCase())
    )
    
    const targetNode = nodes.find(n => 
      n.id !== sourceNode?.id && 
      n.id !== excludeId &&
      (messageLower.includes(n.label.toLowerCase()) || 
       (suggestionNodeLabel && suggestionNodeLabel.toLowerCase() === n.label.toLowerCase()))
    )

    return { source: sourceNode, target: targetNode }
  }

  static findRelationshipById(relationships: Relationship[], id: string): Relationship | undefined {
    return relationships.find(r => r.id === id)
  }

  static findRelationshipByCriteria(
    relationships: Relationship[],
    type: string,
    from: string,
    to: string
  ): Relationship | undefined {
    return relationships.find(r => 
      r.type === type &&
      r.from === from &&
      r.to === to
    )
  }

  static sortRelationshipsByTimestamp(relationships: Relationship[]): Relationship[] {
    return [...relationships].sort((a, b) => {
      const aMatch = a.id.match(/rel_(\d+)_/)
      const bMatch = b.id.match(/rel_(\d+)_/)
      if (aMatch && bMatch) {
        return parseInt(bMatch[1]) - parseInt(aMatch[1])
      }
      return 0
    })
  }
}
```

## External Services

### TIB Terminology Service

Integrates with the Terminology Service for semantic enrichment:

```typescript
import type { TibOntology, TibClass, TibProperty, TibApiResponse } from '../types/semanticTypes'

// Simple LRU Cache implementation
class LRUCache<T> {
    private cache: Map<string, { value: T; timestamp: number }>
    private maxSize: number
    private ttl: number // Time to live in milliseconds

    constructor(maxSize: number = 100, ttlMinutes: number = 30) {
        this.cache = new Map()
        this.maxSize = maxSize
        this.ttl = ttlMinutes * 60 * 1000
    }

    get(key: string): T | null {
        const item = this.cache.get(key)
        if (!item) return null

        // Check if expired
        if (Date.now() - item.timestamp > this.ttl) {
            this.cache.delete(key)
            return null
        }

        // Move to end (LRU)
        this.cache.delete(key)
        this.cache.set(key, item)
        return item.value
    }

    set(key: string, value: T): void {
        // Remove oldest if at capacity
        if (this.cache.size >= this.maxSize) {
            const firstKey = this.cache.keys().next().value
            if (firstKey !== undefined) {
                this.cache.delete(firstKey)
            }
        }

        this.cache.set(key, { value, timestamp: Date.now() })
    }

    clear(): void {
        this.cache.clear()
    }
}

// Cache instances
const ontologiesCache = new LRUCache<TibApiResponse<TibOntology>>(50, 30)
const classesCache = new LRUCache<TibApiResponse<TibClass>>(100, 30)
const propertiesCache = new LRUCache<TibApiResponse<TibProperty>>(100, 30)

// TIB Terminology Service functions
export async function searchOntologies(query: string): Promise<TibApiResponse<TibOntology>> {
    const cacheKey = `search-ontologies:${query}`
    const cached = ontologiesCache.get(cacheKey)
    if (cached) return cached

    try {
        const response = await fetch(`https://service.tib.eu/ts4tib2/api/search?q=${encodeURIComponent(query)}&type=ontology`)
        const data = await response.json()
        
        ontologiesCache.set(cacheKey, data)
        return data
    } catch (error) {
        throw new Error(`Failed to search ontologies: ${error.message}`)
    }
}

export async function getOntology(id: string): Promise<TibOntology | null> {
    const cacheKey = `ontology:${id}`
    const cached = ontologiesCache.get(cacheKey)
    if (cached) return cached.response[0] || null

    try {
        const response = await fetch(`https://service.tib.eu/ts4tib2/api/v2/ontologies/${id}`)
        const data = await response.json()
        
        ontologiesCache.set(cacheKey, { response: [data] })
        return data
    } catch (error) {
        throw new Error(`Failed to get ontology: ${error.message}`)
    }
}

export async function searchClasses(query: string, ontologyId?: string): Promise<TibApiResponse<TibClass>> {
    const cacheKey = `search-classes:${query}:${ontologyId || 'all'}`
    const cached = classesCache.get(cacheKey)
    if (cached) return cached

    try {
        const url = ontologyId 
            ? `https://service.tib.eu/ts4tib2/api/search?q=${encodeURIComponent(query)}&type=class&ontology=${ontologyId}`
            : `https://service.tib.eu/ts4tib2/api/search?q=${encodeURIComponent(query)}&type=class`
            
        const response = await fetch(url)
        const data = await response.json()
        
        classesCache.set(cacheKey, data)
        return data
    } catch (error) {
        throw new Error(`Failed to search classes: ${error.message}`)
    }
}
```

```typescript
 * Handles fetching data from various research APIs like Wikidata, GND, VIAF, etc.

export type ApiProvider = 
  | 'wikidata'
  | 'gnd'
  | 'viaf'
  | 'orcid'
  | 'geonames'
  | 'dblp'
  | 'crossref'
  | 'europeana'
  | 'getty'
  | 'loc'
  | 'custom'

export interface ApiConfig {
  provider: ApiProvider
  id: string
  apiKey?: string
  credentialId?: string // ID of stored credential
  customEndpoint?: string
  customHeaders?: Record<string, string>
  timeout?: number
}

export interface ApiResponse {
  success: boolean
  data?: unknown
  error?: string
  provider: ApiProvider
}

/**
 * Build API endpoint URL based on provider and ID
 */
function buildApiUrl(provider: ApiProvider, id: string, apiKey?: string): string {
  const cleanId = id.trim()
  
  switch (provider) {
    case 'wikidata': {
      // Wikidata Entity API - IDs are case-sensitive and start with Q or P
      const normalizedId = cleanId.match(/^[qp]\d+$/i) ? cleanId.toUpperCase() : cleanId
      return `https://www.wikidata.org/wiki/Special:EntityData/${normalizedId}.json`
    }
    
    case 'gnd':
      // GND SRU API (simplified - may need more complex query building)
      return `https://lobid.org/gnd/search?q=${encodeURIComponent(cleanId)}&format=json`
    
    case 'viaf':
      // VIAF API
      return `https://viaf.org/viaf/${cleanId}`
    
    case 'orcid':
      // ORCID Public API
      return `https://pub.orcid.org/v3.0/${cleanId}`
    
    case 'geonames':
      // GeoNames API
      return `http://api.geonames.org/get?geonameId=${cleanId}&username=demo`
    
    case 'dblp':
      // DBLP API
      return `https://dblp.org/search/publ/api?q=${encodeURIComponent(cleanId)}&format=json`
    
    case 'crossref':
      // Crossref API
      return `https://api.crossref.org/works/${cleanId}`
    
    case 'europeana':
      // Europeana API
      return `https://api.europeana.eu/record/v2/${cleanId}.json?wskey=${apiKey || 'demo'}`
    
    case 'getty':
      // Getty Vocabulary API
      return `http://vocab.getty.edu/sparql.json?query=${encodeURIComponent(`SELECT * WHERE { <http://vocab.getty.edu/tgn/${cleanId}> ?p ?o }`)}`
    
    case 'loc':
      // Library of Congress API
      return `https://id.loc.gov/authorities/${cleanId}.json`
    
    case 'custom':
      // Custom endpoint - should be provided in config
      throw new Error('Custom endpoint must be specified in config.customEndpoint')
    
    default:
      throw new Error(`Unsupported API provider: ${provider}`)
  }
}

/**
 * Fetch data from external API
 */
export async function fetchFromAPI(config: ApiConfig): Promise<ApiResponse> {
  try {
    const url = config.customEndpoint || buildApiUrl(config.provider, config.id, config.apiKey)
    
    const headers: Record<string, string> = {
      'Accept': 'application/json',
      'User-Agent': 'Plexus-Builder/1.0',
      ...config.customHeaders
    }

    // Add API key to headers for certain providers
    if (config.apiKey) {
      switch (config.provider) {
        case 'orcid':
          headers['Authorization'] = `Bearer ${config.apiKey}`
          break
        case 'europeana':
          // API key is already in URL for Europeana
          break
        case 'geonames':
          // API key is in URL parameter for GeoNames
          break
      }
    }

    const controller = new AbortController()
    const timeoutId = setTimeout(() => controller.abort(), config.timeout || 10000)

    const response = await fetch(url, {
      method: 'GET',
      headers,
      signal: controller.signal
    })

    clearTimeout(timeoutId)

    if (!response.ok) {
      throw new Error(`HTTP ${response.status}: ${response.statusText}`)
    }

    const data = await response.json()

    return {
      success: true,
      data,
      provider: config.provider
    }

  } catch (error) {
    return {
      success: false,
      error: error instanceof Error ? error.message : 'Unknown error',
      provider: config.provider
    }
  }
}

/**
 * Search across multiple providers
 */
export async function searchAcrossProviders(
  query: string,
  providers: ApiProvider[],
  credentials?: Record<string, string>
): Promise<ApiResponse[]> {
  const results = await Promise.allSettled(
    providers.map(provider => 
      fetchFromAPI({
        provider,
        id: query,
        apiKey: credentials?.[provider],
        timeout: 5000
      })
    )
  )

  return results.map(result => 
    result.status === 'fulfilled' ? result.value : {
      success: false,
      error: result.reason instanceof Error ? result.reason.message : 'Unknown error',
      provider: 'custom' as ApiProvider
    }
  )
}
```

## Available Services

The Plexus Builder services layer includes the following key services:

### Core Services
- **Workflow Executor**: Executes XML-to-graph transformation workflows (`executeWorkflow`)
- **XML Services**: Parse CSV data and analyze XML structure (`parseCSVLine`, `parseBulkNodeInput`, `XmlAnalyzer`)
- **Schema Services**: Import schema files and provide design utilities (`ImportService`, `SchemaDesignService`)
- **API Client**: Generic client for external research APIs (`fetchFromAPI`, `buildApiUrl`)

### External Services
- **TIB Terminology Service**: Semantic enrichment through terminology lookup (`searchOntologies`, `getOntology`, `searchClasses`)
- **Research APIs**: Integration with Wikidata, GND, VIAF, ORCID, GeoNames, DBLP, Crossref, Europeana, Getty, LOC

### Utility Services
- **LRU Cache**: Performance optimization through intelligent caching (used by TIB service)
- **Validators**: Data validation for workflows, schemas, and mappings
