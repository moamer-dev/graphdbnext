# State Management

Plexus Platform uses a dual state management approach: Zustand for client-side state and TanStack Query for server state. This combination provides optimal performance, type safety, and developer experience.

#### Async Action Pattern
```typescript
// Async action wrapper for error handling
const withAsyncHandler = <T>(
  store: any,
  actionName: string,
  asyncFn: () => Promise<T>
) => {
  store.setLoading(true);
  store.clearError();
  
  try {
    const result = await asyncFn();
    return result;
  } catch (error) {
    store.setError(error instanceof Error ? error.message : 'Unknown error');
    throw error;
  } finally {
    store.setLoading(false);
  }
};

// Usage in store
const databaseStore = createBaseStore('database-store', {
  // ... initial state
  
  checkStatus: async () => {
    return withAsyncHandler(databaseStore, 'checkStatus', async () => {
      const status = await databaseService.checkStatus();
      databaseStore.setState({ dbStatus: status, isConnected: status.connected });
      return status;
    });
  },
});
```

#### Mutations
```typescript
// Hook: Create
function useCreate (options?: { onSuccess?: (data: T) => void }) {
  const queryClient = useQueryClient()
  const router = useRouter()

  return useMutation({
    mutationFn: createItem,
    onSuccess: (data) => {
      // Handle different response formats
      let item: T | undefined
      if ('data' in data) {
        item = (data as SingleResourceResponse<T>).data
      } else {
        const dataRecord = data as unknown as Record<string, T>
        item = dataRecord[config.resourceName.toLowerCase()]
      }
      
      // Invalidate and refetch lists
      queryClient.invalidateQueries({ queryKey: config.queryKeys.lists() })
      
      // Optionally prefetch the new item
      if (item?.id) {
        queryClient.setQueryData(config.queryKeys.detail(item.id), data)
      }
      
      toast.success(`${config.resourceName} created successfully`)
      
      // Navigate to the new item if viewPath is configured
      if (viewPath && item?.id) {
        router.push(`${viewPath}/${item.id}`)
      }
      
      if (item) {
        options?.onSuccess?.(item)
      }
    },
    onError: (error: Error) => {
      toast.error(error.message || `Failed to create ${config.resourceName}`)
    }
  })
}

// Hook: Update
function useUpdate () {
  const queryClient = useQueryClient()

  return useMutation({
    mutationFn: ({ id, data }: { id: string; data: Partial<T> }) => 
      updateItem(id, data),
    onSuccess: (data, variables) => {
      // Update the cache with the new data
      queryClient.setQueryData(config.queryKeys.detail(variables.id), data)
      
      // Invalidate list queries to refetch
      queryClient.invalidateQueries({ queryKey: config.queryKeys.lists() })
      
      toast.success(`${config.resourceName} updated successfully`)
    },
    onError: (error: Error) => {
      toast.error(error.message || `Failed to update ${config.resourceName}`)
    }
  })
}

// Hook: Delete
function useDelete () {
  const queryClient = useQueryClient()
  const router = useRouter()

  return useMutation({
    mutationFn: deleteItem,
    onSuccess: () => {
      // Invalidate list queries
      queryClient.invalidateQueries({ queryKey: config.queryKeys.lists() })
      
      toast.success(`${config.resourceName} deleted successfully`)
      
      // Navigate back to list if listPath is configured
      if (listPath) {
        router.push(listPath)
      }
    },
    onError: (error: Error) => {
      toast.error(error.message || `Failed to delete ${config.resourceName}`)
    }
  })
}
```

## Performance Optimization

### Actual Performance Patterns

#### useMemo for Expensive Calculations
```typescript
// From useQueryResults.ts - Memoize expensive operations
const allKeys = useMemo(() => {
  if (!results || results.length === 0) return []
  const keys = new Set<string>()
  results.forEach((result: unknown) => {
    if (result && typeof result === 'object') {
      Object.keys(result).forEach(key => keys.add(key))
    }
  })
  return Array.from(keys)
}, [results])

// Memoize filtering logic
const filteredResults = useMemo(() => {
  if (!results || results.length === 0) return []
  return results.filter((result: unknown) => {
    if (!result || typeof result !== 'object') return false
    const resultObj = result as Record<string, unknown>

    if (searchTerm) {
      const searchLower = searchTerm.toLowerCase()
      const matchesSearch = Object.values(resultObj).some(value => {
        const searchableText = getSearchableText(value)
        return searchableText.includes(searchLower)
      })
      if (!matchesSearch) return false
    }

    for (const [column, filterValue] of Object.entries(columnFilters)) {
      if (!filterValue) continue
      const value = resultObj[column]
      if (value === null || value === undefined) return false
      
      const searchableText = getSearchableText(value)
      if (!searchableText.includes(filterValue.toLowerCase())) {
        return false
      }
    }

    return true
  })
}, [results, searchTerm, columnFilters])
```

### Key Performance Principles

1. **Memoize Expensive Operations**: Use `useMemo` for filtering, mapping, and calculations
2. **Stable Event Handlers**: Use `useCallback` for event handlers passed to children
3. **Query Caching**: Leverage TanStack Query's built-in caching with appropriate `staleTime`
4. **Placeholder Data**: Use `keepPreviousData` to prevent UI flickering during refetches
5. **Conditional Queries**: Use `enabled` option to prevent unnecessary requests

```typescript
async function fetchList (params: FetchParams): Promise<ResourceResponse<T>> {
  const response = await fetch(`${config.basePath}?${searchParams.toString()}`)
```

## Summary

### State Management Architecture in Plexus Platform

The Plexus Platform uses a clean separation of concerns for state management:

- **Zustand**: Client-side UI state (pagination, search, filters, UI preferences)
- **TanStack Query**: Server state (API data, caching, mutations)
- **No Complex Integration**: They work together naturally at the component level

### Key Benefits

1. **Type Safety**: Full TypeScript support throughout
2. **Performance**: Built-in caching and memoization
3. **Developer Experience**: Simple, predictable APIs
4. **Maintainability**: Clear separation of responsibilities
5. **User Experience**: Smooth interactions with proper loading and error states

### Best Practices Used

1. **Hierarchical Query Keys**: Organized cache structure
2. **Generic Resource Hooks**: Reusable CRUD patterns
3. **Toast Notifications**: Consistent user feedback
4. **Memoization**: Optimize expensive calculations
5. **Error Boundaries**: Graceful error handling

---

This state management architecture provides a robust, performant, and maintainable solution for managing both client and server state in the Plexus Platform, with proper separation of concerns and comprehensive error handling.
