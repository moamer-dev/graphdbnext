Plexus Platform employs various design patterns to ensure maintainability, scalability, and code reusability. This guide covers the key patterns used throughout the application based on the actual implementation.

## Creational Patterns

### Service Factory Pattern

#### Resource Hook Factory
```typescript
// From hooks/react-query/useResource.ts - Generic resource hook factory
export function createResourceHooks<T extends { id: string }>(config: ResourceConfig) {
  const viewPath = config.viewPath || config.listPath
  const listPath = config.listPath || config.basePath.replace('/api/', '/dashboard/')

  // Hook: Create
  function useCreate (options?: { onSuccess?: (data: T) => void }) {
    const queryClient = useQueryClient()
    const router = useRouter()

    return useMutation({
      mutationFn: createItem,
      onSuccess: (data) => {
        // Handle different response formats
        let item: T | undefined
        if ('data' in data) {
          item = (data as SingleResourceResponse<T>).data
        } else {
          const dataRecord = data as unknown as Record<string, T>
          item = dataRecord[config.resourceName.toLowerCase()]
        }
        
        // Invalidate and refetch lists
        queryClient.invalidateQueries({ queryKey: config.queryKeys.lists() })
        
        // Optionally prefetch the new item
        if (item?.id) {
          queryClient.setQueryData(config.queryKeys.detail(item.id), data)
        }
        
        toast.success(`${config.resourceName} created successfully`)
        
        // Navigate to the new item if viewPath is configured
        if (viewPath && item?.id) {
          router.push(`${viewPath}/${item.id}`)
        }
        
        if (item) {
          options?.onSuccess?.(item)
        }
      },
      onError: (error: Error) => {
        toast.error(error.message || `Failed to create ${config.resourceName}`)
      }
    })
  }

  // Hook: Update
  function useUpdate () {
    const queryClient = useQueryClient()

    return useMutation({
      mutationFn: ({ id, data }: { id: string; data: Partial<T> }) => 
        updateItem(id, data),
      onSuccess: (data, variables) => {
        // Update the cache with the new data
        queryClient.setQueryData(config.queryKeys.detail(variables.id), data)
        
        // Invalidate list queries to refetch
        queryClient.invalidateQueries({ queryKey: config.queryKeys.lists() })
        
        toast.success(`${config.resourceName} updated successfully`)
      },
      onError: (error: Error) => {
        toast.error(error.message || `Failed to update ${config.resourceName}`)
      }
    })
  }

  // Hook: Delete
  function useDelete () {
    const queryClient = useQueryClient()
    const router = useRouter()

    return useMutation({
      mutationFn: deleteItem,
      onSuccess: () => {
        // Invalidate list queries
        queryClient.invalidateQueries({ queryKey: config.queryKeys.lists() })
        
        toast.success(`${config.resourceName} deleted successfully`)
        
        // Navigate back to list if listPath is configured
        if (listPath) {
          router.push(listPath)
        }
      },
      onError: (error: Error) => {
        toast.error(error.message || `Failed to delete ${config.resourceName}`)
      }
    })
  }

  return {
    useList,
    useSingle,
    useCreate,
    useUpdate,
    useDelete
  }
}
```

#### Query Builder Service

```typescript
// From services/graph/QueryBuilderService.ts - Cypher query builder
export interface QueryNode {
  id: string
  label?: string
  properties?: Record<string, unknown>
  alias?: string
}

export interface QueryRelationship {
  id: string
  type: string
  source: string
  target: string
  properties?: Record<string, unknown>
  direction?: 'outgoing' | 'incoming' | 'undirected'
}

export interface QueryCondition {
  id: string
  nodeId?: string
  relationshipId?: string
  property: string
  operator: string
  value: unknown
}

export class QueryBuilderService {
  /**
   * Build Cypher query from components
   */
  buildQuery(
    nodes: QueryNode[],
    relationships: QueryRelationship[],
    conditions: QueryCondition[],
    returnFields: string[],
    limit?: number
  ): string {
    const matchClause = this.buildMatchClause(nodes, relationships)
    const whereClause = this.buildWhereClause(conditions)
    const returnClause = this.buildReturnClause(returnFields)
    const limitClause = limit ? ` LIMIT ${limit}` : ''

    return `${matchClause}${whereClause}${returnClause}${limitClause}`.trim()
  }

  private buildMatchClause(nodes: QueryNode[], relationships: QueryRelationship[]): string {
    if (nodes.length === 0) return ''

    let matchClause = 'MATCH '

    // Build node patterns
    const nodePatterns = nodes.map(node => {
      const alias = node.alias || node.id
      const label = node.label ? `:${node.label}` : ''
      const props = node.properties ? ` ${JSON.stringify(node.properties)}` : ''
      return `(${alias}${label}${props})`
    })

    matchClause += nodePatterns.join(', ')

    // Build relationship patterns
    relationships.forEach(rel => {
      const sourceNode = nodes.find(n => n.id === rel.source)
      const targetNode = nodes.find(n => n.id === rel.target)
      
      if (sourceNode && targetNode) {
        const sourceAlias = sourceNode.alias || sourceNode.id
        const targetAlias = targetNode.alias || targetNode.id
        const relProps = rel.properties ? ` ${JSON.stringify(rel.properties)}` : ''
        
        let relPattern = ''
        if (rel.direction === 'incoming') {
          relPattern = `<-[${rel.type}${relProps}]-`
        } else if (rel.direction === 'undirected') {
          relPattern = `-[${rel.type}${relProps}]-`
        } else {
          relPattern = `-[${rel.type}${relProps}]->`
        }
        
        matchClause += `(${sourceAlias})${relPattern}(${targetAlias})`
      }
    })

    return matchClause
  }

  private buildWhereClause(conditions: QueryCondition[]): string {
    if (conditions.length === 0) return ''

    const conditionStrings = conditions.map(condition => {
      let entity = ''
      if (condition.nodeId) {
        entity = condition.nodeId
      } else if (condition.relationshipId) {
        entity = condition.relationshipId
      }

      const operator = this.mapOperator(condition.operator)
      const value = typeof condition.value === 'string' ? `'${condition.value}'` : condition.value
      
      return `${entity}.${condition.property} ${operator} ${value}`
    })

    return ' WHERE ' + conditionStrings.join(' AND ')
  }

  private buildReturnClause(returnFields: string[]): string {
    if (returnFields.length === 0) return ' RETURN *'
    
    return ' RETURN ' + returnFields.join(', ')
  }

  private mapOperator(operator: string): string {
    const operatorMap: Record<string, string> = {
      'equals': '=',
      'not_equals': '!=',
      'greater_than': '>',
      'less_than': '<',
      'greater_equal': '>=',
      'less_equal': '<=',
      'contains': 'CONTAINS',
      'starts_with': 'STARTS WITH',
      'ends_with': 'ENDS WITH'
    }
    
    return operatorMap[operator] || '='
  }
}
```

```typescript
// From services/core/ModuleService.ts - Module management
export interface ModuleConfig {
  name: string
  enabled: boolean
  dependencies?: string[]
  settings?: Record<string, unknown>
}

export class ModuleService {
  private modules: Map<string, ModuleConfig> = new Map()

  /**
   * Register a module
   */
  registerModule(config: ModuleConfig): void {
    this.modules.set(config.name, config)
  }

  /**
   * Enable/disable a module
   */
  setModuleEnabled(name: string, enabled: boolean): void {
    const module = this.modules.get(name)
    if (module) {
      module.enabled = enabled
    }
  }

  /**
   * Check if module is enabled
   */
  isModuleEnabled(name: string): boolean {
    const module = this.modules.get(name)
    return module?.enabled || false
  }

  /**
   * Get all enabled modules
   */
  getEnabledModules(): ModuleConfig[] {
    return Array.from(this.modules.values()).filter(m => m.enabled)
  }
}
```

### Repository Pattern

#### CRUD Service Repository
```typescript
// From services/core/CrudService.ts - Generic CRUD operations
export interface CrudService<T, CreateData, UpdateData> {
  create(data: CreateData): Promise<T>
  findById(id: string): Promise<T | null>
  findMany(filters?: any): Promise<T[]>
  update(id: string, data: UpdateData): Promise<T>
  delete(id: string): Promise<void>
}

export function createCrudService<T, CreateData, UpdateData>(
  config: CrudServiceConfig<T>
): CrudService<T, CreateData, UpdateData> {
  return {
    async create(data: CreateData): Promise<T> {
      const response = await fetch(config.basePath, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data)
      })
      
      if (!response.ok) {
        const error = await response.json().catch(() => ({ error: `Failed to create ${config.resourceName}` }))
        throw new Error(error.error || `Failed to create ${config.resourceName}`)
      }

      return response.json()
    },

    async findById(id: string): Promise<T | null> {
      const response = await fetch(`${config.basePath}/${id}`)
      
      if (!response.ok) {
        if (response.status === 404) return null
        throw new Error(`Failed to fetch ${config.resourceName}`)
      }

      return response.json()
    },

    async findMany(filters?: any): Promise<T[]> {
      const searchParams = new URLSearchParams()
      Object.entries(filters || {}).forEach(([key, value]) => {
        if (value !== undefined && value !== null && value !== '') {
          searchParams.append(key, String(value))
        }
      })

      const response = await fetch(`${config.basePath}?${searchParams.toString()}`)
      
      if (!response.ok) {
        throw new Error(`Failed to fetch ${config.resourceName}s`)
      }

      const result = await response.json()
      return result.data || []
    },

    async update(id: string, data: UpdateData): Promise<T> {
      const response = await fetch(`${config.basePath}/${id}`, {
        method: 'PUT',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(data)
      })
      
      if (!response.ok) {
        throw new Error(`Failed to update ${config.resourceName}`)
      }

      return response.json()
    },

    async delete(id: string): Promise<void> {
      const response = await fetch(`${config.basePath}/${id}`, {
        method: 'DELETE'
      })
      
      if (!response.ok) {
        throw new Error(`Failed to delete ${config.resourceName}`)
      }
    }
  }
}
```

#### Visualization Layout Strategy
```typescript
// From services/visualization/CustomVisualizationService.ts - Layout strategy pattern
export type LayoutType = 'force' | 'hierarchical' | 'circular' | 'grid' | 'timeline'

export class CustomVisualizationService {
  /**
   * Apply layout to nodes
   */
  applyLayout(
    nodes: GraphNode[],
    edges: GraphEdge[],
    config?: VisualizationConfig
  ): GraphNode[] {
    const layout = config?.layout || 'force'
    const options = config?.layoutOptions

    switch (layout) {
      case 'hierarchical':
        return this.applyHierarchicalLayout(nodes, edges, options?.hierarchical)
      case 'circular':
        return this.applyCircularLayout(nodes, edges, options?.circular)
      case 'grid':
        return this.applyGridLayout(nodes, edges, options?.grid)
      case 'timeline':
        return this.applyTimelineLayout(nodes, edges, options?.timeline)
      default:
        return nodes
    }
  }

  private applyCircularLayout(
    nodes: GraphNode[],
    edges: GraphEdge[],
    options?: { radius?: number; startAngle?: number }
  ): GraphNode[] {
    const radius = options?.radius || 200
    const startAngle = options?.startAngle || 0
    const centerX = 400
    const centerY = 300

    const angleStep = (2 * Math.PI) / nodes.length

    nodes.forEach((node, i) => {
      const angle = startAngle + i * angleStep
      node.x = centerX + radius * Math.cos(angle)
      node.y = centerY + radius * Math.sin(angle)
    })

    return nodes
  }

  private applyGridLayout(
    nodes: GraphNode[],
    edges: GraphEdge[],
    options?: { rows?: number; cols?: number }
  ): GraphNode[] {
    const cols = options?.cols || Math.ceil(Math.sqrt(nodes.length))
    const rows = options?.rows || Math.ceil(nodes.length / cols)
    const cellWidth = 800 / cols
    const cellHeight = 600 / rows

    nodes.forEach((node, i) => {
      const row = Math.floor(i / cols)
      const col = i % cols
      node.x = col * cellWidth + cellWidth / 2
      node.y = row * cellHeight + cellHeight / 2
    })

    return nodes
  }

  // ... other layout methods
}
```

## Pattern Benefits

### Advantages of Used Patterns

#### Service Factory Pattern
- **Code Reusability**: Generic resource hooks reduce duplication
- **Type Safety**: Strong typing with TypeScript generics
- **Consistency**: Standardized CRUD operations across resources

#### Builder Pattern
- **Fluent Interface**: Easy-to-read query building
- **Flexibility**: Complex queries built step by step
- **Validation**: Query validation at build time

#### Observer Pattern (Zustand)
- **Performance**: Selective subscriptions prevent unnecessary re-renders
- **Simplicity**: Clean state management without boilerplate
- **Type Safety**: Full TypeScript support

#### Strategy Pattern
- **Flexibility**: Easy to switch AI providers or layouts
- **Extensibility**: Simple to add new strategies
- **Testability**: Each strategy can be tested independently

#### Service Isolation Pattern
- **Build Safety**: Client/Server separation prevents build errors
- **Performance**: Client-only code loads faster
- **Security**: Server-only code protected from client access

#### Hook Pattern
- **Encapsulation**: Complex logic encapsulated in reusable hooks
- **Composition**: Easy to combine multiple hooks
- **Testing**: Hooks can be tested independently

#### Repository Pattern
- **Abstraction**: Clean data access interface
- **Testability**: Easy to mock for testing
- **Flexibility**: Can swap implementations without changing calling code

---

These patterns provide a solid foundation for building maintainable, scalable, and testable software in the Plexus Platform, with emphasis on type safety, performance, and developer experience.
